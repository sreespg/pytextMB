% Matches Elsevier's own shipped cas-dc-sample.pdf exactly: that PDF's
% embedded fonts (checked via `mutool info`) are STIXGeneral for text,
% STIXMath for math, SFSS0900/SFSX0900 (Computer/Latin Modern Sans) for
% \sffamily material (captions, ARTICLE INFO/ABSTRACT headers), and
% Inconsolata for monospace -- i.e. charis.sty was not present when
% Elsevier built it, so cas-dc.cls's fallback \RequirePackage{stix}
% (STIX for both text and math, no Charis, default CM/LM sans) is the
% actual reference appearance, not the charis+stix combination the
% class would prefer if both were available. STIX and Inconsolata
% installed locally (fonts-stix, fonts-inconsolata); Latin Modern Sans
% ships with texlive already. fontspec overrides cas-dc.cls's own
% pdfLaTeX-era stix.sty loading regardless. Without STIX or Inconsolata
% installed, the build falls back to TeX Gyre Termes (a Times design, as
% STIX is) and Latin Modern Mono, which TeX Live ships, and warns.
\IfFontExistsTF{STIX}{\setmainfont{STIX}}{%
  \PackageWarningNoLine{pytexMB}{STIX not installed; using TeX Gyre Termes}%
  \setmainfont{TeX Gyre Termes}}
\setsansfont{Latin Modern Sans}
\IfFontExistsTF{Inconsolata}{\setmonofont{Inconsolata}}{%
  \PackageWarningNoLine{pytexMB}{Inconsolata not installed; using Latin Modern Mono}%
  \setmonofont{Latin Modern Mono}}
